    <script>
        function createRadarWaves() {
            setTimeout(() => {
                const disappearingProfile = document.querySelector('.profile-disappear');
                const rect = disappearingProfile.getBoundingClientRect();
                const container = document.querySelector('.container');
                
                // Position de départ (centre du profil qui disparaît)
                const startX = rect.left + rect.width / 2 - container.getBoundingClientRect().left;
                const startY = rect.top + rect.height / 2 - container.getBoundingClientRect().top;
                
                // Créer 3 ondes successives
                for (let i = 0; i < 3; i++) {
                    setTimeout(() => {
                        const wave = document.createElement('div');
                        wave.className = 'radar-wave';
                        wave.style.left = startX + 'px';
                        wave.style.top = startY + 'px';
                        wave.style.transform = 'translate(-50%, -50%)';
                        container.appendChild(wave);
                        
                        // Supprimer l'onde après l'animation
                        setTimeout(() => wave.remove(), 2000);
                    }, i * 400);
                }
            }, 3000); // Démarrer quand le profil commence à disparaître
        }

        function createNotification() {
            setTimeout(() => {
                const phoneScreen = document.querySelector('.phone-screen');
                const notification = document.createElement('div');
                notification.className = 'notification';
                notification.innerHTML = `
                    <div class="notification-header">
                        <div class="notification-app">Flowtrack</div>
                    </div>
                    <div class="notification-body">
                        <strong>Thomas_95</strong> unfollow you on instagram
                    </div>
                `;
                phoneScreen.appendChild(notification);
            }, 4500);
        }

        function replayAnimation() {
            location.reload();
        }

        // Démarrer les animations
        createRadarWaves();
        createNotification();
    </script>
